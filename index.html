-- ============================================================ -- BANCO ÚNICO + PREVISÕES CONGELADAS (PostgreSQL) -- ============================================================ CREATE EXTENSION IF NOT EXISTS pgcrypto; -- 1. ENUMS CREATE TYPE draw_session AS ENUM ('MANHA', 'TARDE', 'NOITE'); CREATE TYPE prediction_status AS ENUM ('OPEN', 'RESOLVED', 'INVALIDATED'); -- 2. HISTÓRICO: EVENTOS (draws) CREATE TABLE draws ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), draw_date DATE NOT NULL, draw_time TIME, event_datetime TIMESTAMPTZ NOT NULL, session draw_session NOT NULL, contest_number BIGINT, source VARCHAR(100), imported_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), UNIQUE (draw_date, session, contest_number) ); CREATE INDEX idx_draws_event_datetime ON draws(event_datetime); CREATE INDEX idx_draws_session_datetime ON draws(session, event_datetime); -- 3. HISTÓRICO: RESULTADOS (draw_results) CREATE TABLE draw_results ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), draw_id UUID NOT NULL REFERENCES draws(id) ON DELETE CASCADE, position SMALLINT NOT NULL, number CHAR(4) NOT NULL, thousands SMALLINT NOT NULL, hundreds SMALLINT NOT NULL, tens SMALLINT NOT NULL, units SMALLINT NOT NULL, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), CONSTRAINT chk_position CHECK (position BETWEEN 1 AND 10), CONSTRAINT chk_number_format CHECK (number ~ '^[0-9]{4}$'), CONSTRAINT chk_digits CHECK ( thousands BETWEEN 0 AND 9 AND hundreds BETWEEN 0 AND 9 AND tens BETWEEN 0 AND 9 AND units BETWEEN 0 AND 9 ), UNIQUE(draw_id, position) ); CREATE INDEX idx_draw_results_draw ON draw_results(draw_id); CREATE INDEX idx_draw_results_number ON draw_results(number); -- 4. INFRAESTRUTURA: IMPORTAÇÕES E MODELOS CREATE TABLE import_jobs ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), filename VARCHAR(500) NOT NULL, file_type VARCHAR(20) NOT NULL, file_hash VARCHAR(128), status VARCHAR(30) NOT NULL DEFAULT 'PENDING', created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), completed_at TIMESTAMPTZ ); CREATE TABLE models ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), name VARCHAR(150) NOT NULL, version VARCHAR(50) NOT NULL, model_type VARCHAR(100) NOT NULL, parameters JSONB NOT NULL DEFAULT '{}'::JSONB, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), UNIQUE(name, version) ); CREATE TABLE backtest_runs ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), model_id UUID REFERENCES models(id) ON DELETE SET NULL, name VARCHAR(200) NOT NULL, started_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), cutoff_strategy VARCHAR(100), parameters JSONB NOT NULL DEFAULT '{}'::JSONB, metrics JSONB NOT NULL DEFAULT '{}'::JSONB, leakage_detected BOOLEAN NOT NULL DEFAULT FALSE, status VARCHAR(30) NOT NULL DEFAULT 'RUNNING', completed_at TIMESTAMPTZ ); -- 5. PREVISÕES CONGELADAS (Cabeçalho) CREATE TABLE prediction_runs ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), model_id UUID REFERENCES models(id) ON DELETE SET NULL, backtest_run_id UUID REFERENCES backtest_runs(id) ON DELETE SET NULL, model_name VARCHAR(150) NOT NULL, model_version VARCHAR(50) NOT NULL, target_draw_id UUID REFERENCES draws(id) ON DELETE SET NULL, target_event_datetime TIMESTAMPTZ NOT NULL, target_session draw_session NOT NULL, cutoff_datetime TIMESTAMPTZ NOT NULL, last_data_datetime TIMESTAMPTZ, parameters JSONB NOT NULL DEFAULT '{}'::JSONB, features JSONB NOT NULL DEFAULT '{}'::JSONB, dataset_hash VARCHAR(128), model_hash VARCHAR(128), status prediction_status NOT NULL DEFAULT 'OPEN', leakage_detected BOOLEAN NOT NULL DEFAULT FALSE, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), resolved_at TIMESTAMPTZ ); CREATE INDEX idx_prediction_runs_target ON prediction_runs(target_event_datetime); CREATE INDEX idx_prediction_runs_cutoff ON prediction_runs(cutoff_datetime); -- 6. PREVISÕES CONGELADAS (Itens / Ranking Imutável) CREATE TABLE frozen_prediction_items ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), prediction_run_id UUID NOT NULL REFERENCES prediction_runs(id) ON DELETE RESTRICT, ranking INTEGER NOT NULL CHECK (ranking > 0), number CHAR(4) NOT NULL CHECK (number ~ '^[0-9]{4}$'), score NUMERIC(18,10), estimated_probability NUMERIC(18,15) CHECK (estimated_probability IS NULL OR (estimated_probability >= 0 AND estimated_probability <= 1)), actual_hit BOOLEAN, actual_position SMALLINT, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), UNIQUE(prediction_run_id, ranking), UNIQUE(prediction_run_id, number) ); -- 7. RECONCILIAÇÃO / RESULTADOS DA PREVISÃO CREATE TABLE prediction_results ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), prediction_run_id UUID NOT NULL REFERENCES prediction_runs(id) ON DELETE RESTRICT, evaluated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), top_1_hit BOOLEAN, top_10_hit BOOLEAN, total_hits INTEGER DEFAULT 0, UNIQUE(prediction_run_id) ); -- 8. AUDITORIA CREATE TABLE audit_events ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), entity_type VARCHAR(100) NOT NULL, entity_id UUID, action VARCHAR(100) NOT NULL, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW() );
